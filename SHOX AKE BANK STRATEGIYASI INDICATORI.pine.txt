box_disp_mode    = input.string("Barcha Signallar (Tarix)", "Boxlarni Ko'rsatish Rejimi", options=["Barcha Signallar (Tarix)", "Faqat Oxirgi Signal"], group=grp_liquidity, tooltip="Tarixdagi barcha signallarda boxlarni ko'rsatish yoki faqat oxirgisida.")

f_get_pip_size() =>
    float res = syminfo.mintick * 10
    if custom_pip_val > 0.0
        res := custom_pip_val
    else if str.contains(syminfo.ticker, "XAU") or str.contains(syminfo.ticker, "GOLD")
        res := 0.10
    else if str.contains(syminfo.ticker, "JPY")
        res := 0.01
    else if str.contains(syminfo.ticker, "USOIL") or str.contains(syminfo.ticker, "BRENT") or str.contains(syminfo.ticker, "WTI")
        res := 0.01
    else if syminfo.type == "index" or str.contains(syminfo.ticker, "US30") or str.contains(syminfo.ticker, "NAS100") or str.contains(syminfo.ticker, "SPX500")
        res := 1.00
    else if syminfo.type == "forex"
        res := 0.0001
    res

pip_value = f_get_pip_size()

// 3. TREND VA SEANSLAR
// =============================================================================

ema200 = ta.ema(close, ema_len)
plot(show_ema ? ema200 : na, "EMA 200", color=color.new(color.orange, 0), linewidth=2)

trend_bullish = close > ema200
trend_bearish = close < ema200

// Seanslar (UTC)
f_in_session(sess_str) => not na(time(timeframe.period, sess_str + ":1234567", "UTC"))

is_asia   = f_in_session("0000-0800")
is_london = f_in_session("0700-1500")
is_ny     = f_in_session("1200-2000")

current_session = is_london and is_ny ? "London/NY" : is_london ? "London" : is_ny ? "New York" : is_asia ? "Asian" : "Overnight"

// =============================================================================
// 4. BANK MANIPULYATSIYASI (LIQUIDITY SWEEP)
// =============================================================================

// Swing darajalar (Pivot High & Low)
swing_high = ta.pivothigh(high, swing_length, swing_length)
swing_low  = ta.pivotlow(low, swing_length, swing_length)

var float last_swing_high = na
var float last_swing_low  = na

if not na(swing_high)
    last_swing_high := swing_high

if not na(swing_low)
    last_swing_low := swing_low

// Likvidlik ovi (Sweep):
sweep_high = not na(last_swing_high) and high > last_swing_high and close < last_swing_high
sweep_low  = not na(last_swing_low) and low < last_swing_low and close > last_swing_low

// Displacement (Kuchli teskari impuls tasdig'i)
body_size = math.abs(close - open)
avg_body  = ta.sma(body_size, 10)

bullish_displacement = require_displace ? (close > open and body_size > avg_body and close > high[1]) : true
bearish_displacement = require_displace ? (close < open and body_size > avg_body and close < low[1]) : true

// =============================================================================
// 5. SIGNALLARNI HISOBLASH (SHOH AKE STRATEGY)
// =============================================================================

var bool last_sweep_was_low  = false
var bool last_sweep_was_high = false
var int  bars_since_low_sweep  = 999
var int  bars_since_high_sweep = 999

if sweep_low
    last_sweep_was_low   := true
    bars_since_low_sweep := 0
else
    bars_since_low_sweep := bars_since_low_sweep + 1

if sweep_high
    last_sweep_was_high   := true
    bars_since_high_sweep := 0
else
    bars_since_high_sweep := bars_since_high_sweep + 1

// Signal shartlari
raw_buy_signal  = (sweep_low or (bars_since_low_sweep <= 3 and last_sweep_was_low)) and bullish_displacement and (use_ema_filter ? trend_bullish : true)
raw_sell_signal = (sweep_high or (bars_since_high_sweep <= 3 and last_sweep_was_high)) and bearish_displacement and (use_ema_filter ? trend_bearish : true)

// Takroriy signallarni filtrlash
var int last_signal_bar  = -100
var int last_signal_type = 0 // 1: Buy, -1: Sell

buy_signal  = raw_buy_signal and (bar_index - last_signal_bar > 5) and (last_signal_type != 1)
sell_signal = raw_sell_signal and (bar_index - last_signal_bar > 5) and (last_signal_type != -1)

if buy_signal
    last_signal_bar := bar_index
    last_signal_type := 1
    last_sweep_was_low := false

if sell_signal
    last_signal_bar := bar_index
    last_signal_type := -1
    last_sweep_was_high := false

// =============================================================================
// 6. ATR VA 1:1, 1:2, 1:3 STOP LOSS & TAKE PROFIT HISOBLARI
// =============================================================================

atr_val = ta.atr(atr_len)

var float ep   = na
var float sl   = na
var float tp1  = na
var float tp2  = na
var float tp3  = na
var float sl_pips_actual  = na
var float tp1_pips_actual = na
var float tp2_pips_actual = na
var float tp3_pips_actual = na

if buy_signal
    ep := close
    float sl_distance = 0.0
    if sl_mode == "ATR Based"
        sl_distance := atr_val * atr_mult
        sl := ep - sl_distance
    else if sl_mode == "Fixed Pips (Razgon)"
        sl_distance := sl_pips_input * pip_value
        sl := ep - sl_distance
    else
        sl := math.min(low, not na(last_swing_low) ? last_swing_low : low) - (1.0 * pip_value)
        sl_distance := ep - sl
    
    sl_pips_actual  := sl_distance / pip_value
    tp1 := ep + (sl_distance * tp1_rr)
    tp2 := ep + (sl_distance * tp2_rr)
    tp3 := ep + (sl_distance * tp3_rr)
    tp1_pips_actual := (sl_distance * tp1_rr) / pip_value
    tp2_pips_actual := (sl_distance * tp2_rr) / pip_value
    tp3_pips_actual := (sl_distance * tp3_rr) / pip_value

if sell_signal
    ep := close
    float sl_distance = 0.0
    if sl_mode == "ATR Based"
        sl_distance := atr_val * atr_mult
        sl := ep + sl_distance
    else if sl_mode == "Fixed Pips (Razgon)"
        sl_distance := sl_pips_input * pip_value
        sl := ep + sl_distance
    else
        sl := math.max(high, not na(last_swing_high) ? last_swing_high : high) + (1.0 * pip_value)
        sl_distance := sl - ep
        
    sl_pips_actual  := sl_distance / pip_value
    tp1 := ep - (sl_distance * tp1_rr)
    tp2 := ep - (sl_distance * tp2_rr)
    tp3 := ep - (sl_distance * tp3_rr)
    tp1_pips_actual := (sl_distance * tp1_rr) / pip_value
    tp2_pips_actual := (sl_distance * tp2_rr) / pip_value
    tp3_pips_actual := (sl_distance * tp3_rr) / pip_value

        label.delete(lbl_entry)
        label.delete(lbl_sl)
        label.delete(lbl_tp1)
        label.delete(lbl_tp2)
        label.delete(lbl_tp3)
    
    int right_bar = bar_index + box_length
    
    // Stop Loss Box
    box_sl := box.new(bar_index, math.max(ep, sl), right_bar, math.min(ep, sl), bgcolor=color.new(color.red, 82), border_color=color.new(color.red, 40), border_width=1)
    
    // Take Profit Boxlari
    if buy_signal
        box_tp1 := box.new(bar_index, tp1, right_bar, ep, bgcolor=color.new(color.green, 90), border_color=color.new(color.green, 60), border_width=1)
        box_tp2 := box.new(bar_index, tp2, right_bar, tp1, bgcolor=color.new(color.green, 85), border_color=color.new(color.green, 50), border_width=1)
        box_tp3 := box.new(bar_index, tp3, right_bar, tp2, bgcolor=color.new(color.green, 80), border_color=color.new(color.green, 40), border_width=1)
    else
        box_tp1 := box.new(bar_index, ep, right_bar, tp1, bgcolor=color.new(color.green, 90), border_color=color.new(color.green, 60), border_width=1)
        box_tp2 := box.new(bar_index, tp1, right_bar, tp2, bgcolor=color.new(color.green, 85), border_color=color.new(color.green, 50), border_width=1)
        box_tp3 := box.new(bar_index, tp2, right_bar, tp3, bgcolor=color.new(color.green, 80), border_color=color.new(color.green, 40), border_width=1)
    
    // Chiziqlar
    line_entry := line.new(bar_index, ep, right_bar, ep, color=color.blue, width=2)

    line_tp1   := line.new(bar_index, tp1, right_bar, tp1, color=color.new(color.green, 20), width=1, style=line.style_dotted)
    line_tp2   := line.new(bar_index, tp2, right_bar, tp2, color=color.new(color.green, 10), width=1, style=line.style_dotted)
    line_tp3   := line.new(bar_index, tp3, right_bar, tp3, color=color.green, width=2, style=line.style_dashed)
    
    // Yorliqlar
    if show_labels
        lbl_entry := label.new(right_bar, ep, "ENTRY: " + str.tostring(ep, "#.##"), style=label.style_label_left, color=color.blue, textcolor=color.white, size=size.small)
        lbl_sl    := label.new(right_bar, sl, "SL: " + str.tostring(sl, "#.##") + " (-" + str.tostring(sl_pips_actual, "#.#") + "p)", style=label.style_label_left, color=color.red, textcolor=color.white, size=size.small)
        lbl_tp1   := label.new(right_bar, tp1, "TP1 (1:1): " + str.tostring(tp1, "#.##") + " (+" + str.tostring(tp1_pips_actual, "#.#") + "p)", style=label.style_label_left, color=color.green, textcolor=color.white, size=size.small)
        lbl_tp2   := label.new(right_bar, tp2, "TP2 (1:2): " + str.tostring(tp2, "#.##") + " (+" + str.tostring(tp2_pips_actual, "#.#") + "p)", style=label.style_label_left, color=color.green, textcolor=color.white, size=size.small)
        lbl_tp3   := label.new(right_bar, tp3, "TP3 (1:3): " + str.tostring(tp3, "#.##") + " (+" + str.tostring(tp3_pips_actual, "#.#") + "p)", style=label.style_label_left, color=#00e676, textcolor=color.black, size=size.small)
